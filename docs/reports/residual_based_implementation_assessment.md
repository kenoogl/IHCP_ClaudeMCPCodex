    ...

        res /= res0  # 相対残差

        if res < tol

### 3. AdaptiveToleranceモジュールの拡張

#### 新規関数の追加

```julia
"""
残差履歴ベースの適応的収束判定

# アルゴリズム
tolₙ = clamp( κ_r · ρ̄_{n-1}^α, tol_min, tol_max )

# 引数
- residual_history: 前ステップまでの最終残差（移動平均用）
- κ_r: スケール係数（推奨: 0.5）
- α: べき乗係数（推奨: 0.7）
"""
function compute_adaptive_tol_residual_based(
    residual_history::Vector{T},
    t::Int,
    tol_default::T,
    params::AdaptiveToleranceResidualParams{T}
) where {T <: AbstractFloat}
    # 初期ステップは固定tol使用
    if t <= params.warmup_steps
        return tol_default
    end

    # 移動平均の計算
    window_size = min(params.window_size, length(residual_history))
    ρ_avg = mean(residual_history[end-window_size+1:end])

    # 適応的tol計算
    tol_adaptive = params.κ_r * ρ_avg^params.α

    # クランプ
    return clamp(tol_adaptive, params.tol_min, params.tol_max)
end
```

**実装難易度**: 🟡 **中**（新規関数、パラメータ構造体の追加）
